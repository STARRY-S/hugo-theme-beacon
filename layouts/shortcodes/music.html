Usage: {{< music netease song 594295 >}} positional: server type id {{< music server="netease" type="playlist" id="8433061690" >}} {{< music auto="https://music.163.com/#/playlist?id=…" >}} {{< music url="song.mp3" name="…" artist="…" cover="cover.jpg" >}} url/cover accept a page-bundle resource path or an absolute URL. */ -}} {{- $theme := "#0066d6" -}} {{- if .IsNamedParams -}} {{- $theme = .Get "theme" | default $theme -}} {{- if not (findRE `^#[0-9A-Fa-f]{6}$` $theme) -}} {{- errorf "music shortcode: theme must use #RRGGBB; see %s" .Position -}} {{- end -}} {{- if .Get "url" -}} {{- $url := .Get "url" -}} {{- with .Page.Resources.Get $url }}{{ $url = .RelPermalink }}{{ end -}} {{- $cover := .Get "cover" -}} {{- with .Page.Resources.Get $cover }}{{ $cover = .RelPermalink }}{{ end -}} {{- if and $cover (not (or (strings.Contains $cover "://") (hasPrefix $cover "/"))) }}{{ $cover = partial "asset-url.html" $cover }}{{ end -}} {{- if and (not (strings.Contains $url "://")) (not (hasPrefix $url "/")) }}{{ $url = partial "asset-url.html" $url }}{{ end -}} {{- else if .Get "auto" -}} {{- else -}} {{- if or (not (.Get "server")) (not (.Get "type")) (not (.Get "id")) -}} {{- errorf "music shortcode: named usage requires server, type, and id; see %s" .Position -}} {{- end -}} {{- end -}} {{- else if lt (len .Params) 1 -}} {{- errorf "music shortcode: provide an auto URL or server, type, and id; see %s" .Position -}} {{- else if (.Get 0 | urls.Parse).Host -}} {{- else -}} {{- if lt (len .Params) 3 -}}{{ errorf "music shortcode: positional usage requires server, type, and id; see %s" .Position }}{{ end -}} {{- end -}}

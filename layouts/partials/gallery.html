{{- /* Gallery / photo waterfall. Renders a slice of image page-resources as a masonry (CSS columns) grid with click-to-zoom via the shared lightbox (image-preview.js). Thumbnails and a size-capped full image are generated with Hugo image processing — no external service, all fingerprinted and cached. Call with a dict: {{ partial "gallery.html" (dict "images" $images "page" .) }} Keys: images - a slice of image resources (e.g. .Resources.ByType "image"). page - the page (used for error context); optional. thumb - thumbnail width in px (default 800). reverse - reverse the given order (default false). Per-image caption comes from the resource's front-matter params: set `caption` (and optionally `alt`) under [[resources]] in the page front matter; otherwise the image shows no caption. */ -}} {{- $images := .images -}} {{- if .reverse }}{{ $images = collections.Reverse $images }}{{ end -}} {{- $thumbW := .thumb | default 800 -}} {{- $fullW := 2000 -}} {{- if $images -}}
{{- range $images -}} {{- $img := . -}} {{- $thumb := $img -}} {{- if gt $img.Width $thumbW }}{{ $thumb = $img.Resize (printf "%dx q82" $thumbW) }}{{ end -}} {{- $full := $img -}} {{- if gt $img.Width $fullW }}{{ $full = $img.Resize (printf "%dx q85" $fullW) }}{{ end -}} {{- $caption := "" -}} {{- with $img.Params.caption }}{{ $caption = . }}{{ end -}} {{- $alt := or $img.Params.alt $caption "" -}} {{- $exif := partial "gallery-exif.html" $img -}} {{- with $caption }}{{ . | markdownify }}{{ end -}} {{- end -}}

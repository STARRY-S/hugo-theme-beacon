{{- /* Gallery / photo waterfall. Renders a slice of image page-resources as a masonry (CSS columns) grid with click-to-zoom via the shared lightbox (image-preview.js). Thumbnails and a size-capped full image are generated with Hugo image processing — no external service, all fingerprinted and cached. Call with a dict: {{ partial "gallery.html" (dict "images" $images "page" .) }} Keys: images - a slice of image resources (e.g. .Resources.ByType "image"). page - the page the images belong to; needed to find sidecars. thumb - thumbnail width in px (default 800). reverse - reverse the given order (default false). Per-image caption, first hit wins: 1. `caption` under [[resources]] in the page front matter (with `alt`), 2. the Title in a sidecar JSON next to the image (see gallery-meta.html), 3. none. */ -}} {{- $images := .images -}} {{- $page := .page -}} {{- if .reverse }}{{ $images = collections.Reverse $images }}{{ end -}} {{- $thumbW := .thumb | default 800 -}} {{- $fullW := 2000 -}} {{- if $images -}}
{{- range $images -}} {{- $img := . -}} {{- $thumb := $img -}} {{- if gt $img.Width $thumbW }}{{ $thumb = $img.Resize (printf "%dx q82" $thumbW) }}{{ end -}} {{- $full := $img -}} {{- if gt $img.Width $fullW }}{{ $full = $img.Resize (printf "%dx q85" $fullW) }}{{ end -}} {{- $caption := "" -}} {{- with $img.Params.caption }}{{ $caption = . }}{{ end -}} {{- /* Fall back to a sidecar JSON's Title. Front matter wins: it is the explicit, in-repo choice. */ -}} {{- if not $caption }}{{ $caption = partial "gallery-meta.html" (dict "img" $img "page" $page) }}{{ end -}} {{- $alt := or $img.Params.alt $caption "" -}} {{- $exif := partial "gallery-exif.html" $img -}} {{- with $caption }}{{ . | markdownify }}{{ end -}} {{- end -}}
{{- end -}}

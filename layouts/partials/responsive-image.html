{{- $raw := .src -}} {{- $page := .page -}} {{- $resource := false -}} {{- if reflect.IsImageResource $raw -}} {{- $resource = $raw -}} {{- else -}} {{- $path := strings.TrimPrefix "/" (string $raw) -}} {{- if and $page (not (strings.Contains $path "://")) -}} {{- with $page.Resources.GetMatch $path }}{{ $resource = . }}{{ end -}} {{- end -}} {{- if not $resource -}} {{- with resources.Get $path }}{{ $resource = . }}{{ end -}} {{- end -}} {{- end -}} {{- $alt := .alt | default "" -}} {{- $class := .class | default "" -}} {{- $loading := .loading | default "lazy" -}} {{- $sizes := .sizes | default "100vw" -}} {{- $processable := and $resource (partial "image-processable.html" $resource) -}} {{- if $processable -}} {{- $widths := .widths | default (slice 480 800 1200) -}} {{- $targetWidths := slice -}} {{- $webps := slice -}} {{- range $widths -}} {{- if ge $resource.Width . }}{{ $targetWidths = $targetWidths | append . }}{{ end -}} {{- end -}} {{- if not $targetWidths }}{{ $targetWidths = slice $resource.Width }}{{ end -}} {{- range $targetWidths -}} {{- $spec := cond (eq . $resource.Width) "webp q82" (printf "resize %dx webp q82" .) -}} {{- $filters := slice images.AutoOrient (images.Process $spec) -}} {{- $webps = $webps | append ($resource.Filter $filters) -}} {{- end -}} {{- $fallbackWidth := index $targetWidths (sub (len $targetWidths) 1) -}} {{- $fallback := $resource -}} {{- if eq $fallbackWidth $resource.Width -}} {{- $fallback = $resource.Filter images.AutoOrient -}} {{- else -}} {{- $filters := slice images.AutoOrient (images.Process (printf "resize %dx q82" $fallbackWidth)) -}} {{- $fallback = $resource.Filter $filters -}} {{- end -}} {{- else -}} {{- end -}}
